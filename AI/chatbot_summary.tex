\section{How to Run}
% ============================================================

\subsection{Step 1: Install Dependencies}

\begin{lstlisting}[language=bash]
pip install flask flask-cors tensorflow nltk
\end{lstlisting}

\subsection{Step 2: Start the Flask Chatbot API}

\begin{lstlisting}[language=bash]
python chatbot_api.py
\end{lstlisting}

\subsection{Step 3: Integrate with React (MERN Frontend)}

Copy \file{TrivaChatbot.jsx} into your React project's components folder, then add it anywhere in your app:

\begin{lstlisting}[language=Java, caption={App.jsx --- Adding Triva to MERN frontend}]
// In App.jsx or any layout component
import TrivaChatbot from './components/TrivaChatbot';

function App() {
  return (
    <div>
      {/* Your existing MERN app */}
      <TrivaChatbot />   {/* Add this one line */}
    </div>
  );
}
\end{lstlisting}

The chatbot will appear as a floating button (bottom-right corner) on every page of your website.

\section{How the Chatbot Works}
% ============================================================

\subsection{Two-Model Architecture}

\textbf{Model 1 --- General Intent Classifier}
\begin{itemize}[noitemsep]
  \item File: \file{intents.json}
  \item Handles: greetings, goodbyes, general package info, booking info, travel styles
  \item Tags: \tag{greeting}, \tag{goodbye}, \tag{identity}, \tag{destinations}, \tag{budget\_travel}, \tag{adventure\_travel}, etc.
\end{itemize}

\vspace{6pt}
\textbf{Model 2 --- Destination-Specific Responder}
\begin{itemize}[noitemsep]
  \item File: \file{intents1.json}
  \item Handles: specific Indian state/destination details with full package info
  \item Tags: \tag{Goa}, \tag{Kerala}, \tag{Rajasthan}, \tag{Kashmir}, \tag{Andaman}, \tag{Himachal}, \tag{Ladakh}, etc.
\end{itemize}

\subsection{Routing Flow}

\begin{lstlisting}
User Message
     |
     v
Model 1  -->  Classifies intent (tag)
     |
     v
If tag == "destinations"  -->  Model 2  -->  Returns specific package details
Else                       -->  Model 1  -->  Returns general response
\end{lstlisting}
